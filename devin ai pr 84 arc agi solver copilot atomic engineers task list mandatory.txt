    """SHA-256 of grid cells for deterministic identity."""  
    payload = json.dumps(grid.cells, separators=(",", ":"))  
    return hashlib.sha256(payload.encode("utf-8")).hexdigest()[:16]

"""ARC-AGI-3 bounded DSL for program synthesis with proof objects."""  
  
from __future__ import annotations  
  
import itertools  
import zlib  
from typing import Dict, List, Optional, Sequence, Tuple  
  
from axioms.logic import ProofObject  
from axioms.pattern_recognition import (  
    CompositionalRule, Grid, PrimitiveOperation, apply_rule,  
    _candidate_rules_for_pair, _recolor_mapping,  
    infer_conditional_rule, verify_rule,  
)  
  
try:  
    from minimal_ai_ide.maximal_oracle_v57 import KolmogorovComplexityEstimator  
except Exception:  
    class KolmogorovComplexityEstimator:  
        def estimate(self, content: str) -> int:  
            return len(zlib.compress(content.encode("utf-8")))  
  
  
# All single-step operations that can be enumerated without parameters  
PARAMETERLESS_OPS = [  
    PrimitiveOperation.IDENTITY,  
    PrimitiveOperation.ROTATE_90,  
    PrimitiveOperation.ROTATE_180,  
    PrimitiveOperation.ROTATE_270,  
    PrimitiveOperation.REFLECT_H,  
    PrimitiveOperation.REFLECT_V,  
    PrimitiveOperation.DETECT_BOUNDARY,  
    PrimitiveOperation.EXTRACT_OBJECT,  
    PrimitiveOperation.COUNT,  
]  
  
  
def enumerate_single_ops(inp: Grid, out: Grid) -> List[CompositionalRule]:  
    """Enumerate all single-operation rules that transform inp to out."""  
    return _candidate_rules_for_pair(inp, out)  
  
  
def enumerate_compositions(  
    inp: Grid, out: Grid, max_depth: int = 2  
) -> Tuple[List[CompositionalRule], ProofObject]:  
    """  
    Enumerate compositions of parameterless ops up to max_depth.  
    Returns (candidates, proof).  
    """  
    candidates: List[CompositionalRule] = []  
    searched = 0  
  
    for depth in range(1, max_depth + 1):  
        for ops in itertools.product(PARAMETERLESS_OPS, repeat=depth):  
            rule = CompositionalRule([(op, {}) for op in ops])  
            searched += 1  
            if apply_rule(rule, inp) == out:  
                candidates.append(rule)  
  
    # Also try recolor  
    mapping = _recolor_mapping(inp, out)  
    if mapping is not None:  
        rule = CompositionalRule([(PrimitiveOperation.RECOLOR, {"mapping": mapping})])  
        if apply_rule(rule, inp) == out:  
            candidates.append(rule)  
  
    proof = ProofObject(  
        "DSLEnumeration",  
        [f"max_depth={max_depth}", f"searched={searched}",  
         f"candidates_found={len(candidates)}"],  
        f"Enumerated {searched} programs, found {len(candidates)} matches",  
    )  
    return candidates, proof  
  
  
def synthesize_rule(  
    pairs: List[Tuple[Grid, Grid]],  
    max_depth: int = 2,  
    properties: Optional[List] = None,  
) -> Tuple[Optional[CompositionalRule], ProofObject]:  
    """  
    Synthesize a rule that satisfies all training pairs.  
  
    Strategy:  
    1. Try single-op candidates from first pair, verify on all pairs.  
    2. Try depth-N compositions from first pair, verify on all pairs.  
    3. Try conditional rules if signatures diverge.  
    4. Select minimum-complexity candidate (MDL principle).  
  
    Returns (rule_or_None, proof).  
    """  
    if not pairs:  
        return None, ProofObject("Synthesis", ["No pairs"], "No rule synthesized")  
  
    estimator = KolmogorovComplexityEstimator()  
  
    # Phase 1: single-op + recolor from existing _candidate_rules_for_pair  
    first_candidates = _candidate_rules_for_pair(*pairs[0])  
    universal = [c for c in first_candidates  
                 if all(apply_rule(c, inp) == out for inp, out in pairs)]  
  
    # Phase 2: depth-N compositions  
    if not universal:  
        depth_candidates, enum_proof = enumerate_compositions(  
            *pairs[0], max_depth=max_depth  
        )  
        universal = [c for c in depth_candidates  
                     if all(apply_rule(c, inp) == out for inp, out in pairs)]  
  
    # Phase 3: conditional rules  
    if not universal and properties:  
        cond = infer_conditional_rule(pairs, properties)  
        if cond is not None and all(  
            apply_rule(cond, inp) == out for inp, out in pairs  
        ):  
            universal = [cond]  
  
    if not universal:  
        proof = ProofObject(  
            "Synthesis",  
            [f"pairs={len(pairs)}", f"max_depth={max_depth}",  
             f"tried_conditional={properties is not None}"],  
            "No universal rule found",  
        )  
        return None, proof  
  
    # MDL selection: pick minimum Kolmogorov complexity  
    best = min(universal, key=lambda r: r.complexity)  
    proof = ProofObject(  
        "Synthesis",  
        [f"candidates={len(universal)}",  
         f"selected_complexity={best.complexity}",  
         f"ops={[op.value for op, _ in best.operations]}"],  
        f"Synthesized rule with {len(best.operations)} operations",  
    )  
    return best, proof  
  
  
def mdl_compress(  
    rule: CompositionalRule, library: "ConceptLibrary"  
) -> Tuple[List[str], ProofObject]:  
    """  
    Factor a rule into library components via MDL compression.  
    Returns (component_names, proof).  
    """  
    from axioms.arc_types import ConceptLibrary  # avoid circular  
  
    estimator = KolmogorovComplexityEstimator()  
    original_complexity = rule.complexity  
    components_used: List[str] = []  
  
    # Check if any library composition matches a sub-sequence of operations  
    for name, lib_rule in library.compositions.items():  
        lib_ops = [(op.value, params) for op, params in lib_rule.operations]  
        rule_ops = [(op.value, params) for op, params in rule.operations]  
        if str(lib_ops) in str(rule_ops):  
            components_used.append(name)  
  
    compressed_desc = str(components_used) if components_used else str(  
        [(op.value, p) for op, p in rule.operations]  
    )  
    compressed_complexity = estimator.estimate(compressed_desc)  
  
    proof = ProofObject(  
        "MDLCompression",  
        [f"original_complexity={original_complexity}",  
         f"compressed_complexity={compressed_complexity}",  
         f"components_used={components_used}"],  
        f"Compression ratio: {compressed_complexity}/{original_complexity}",  
    )  
    return components_used, proof

def _default_properties() -> list:  
    """Default grid property detectors for conditional rule inference."""  
    return [  
        lambda grid: grid.rows,  
        lambda grid: grid.cols,  
        lambda grid: 1 if grid.rows == grid.cols else 0,  
        lambda grid: sum(1 for row in grid.cells for cell in row if cell != 0),  
        lambda grid: len(grid.get_contiguous_regions()),  
        lambda grid: max(  
            (histogram := grid.get_color_histogram()),  
            key=histogram.get  
        ) if (histogram := grid.get_color_histogram()) else 0,  
        lambda grid: grid.rows * grid.cols,  
        lambda grid: len(set(c for row in grid.cells for c in row)),  
    ]  
  
  
class ARCSolver:  
    """  
    Solves ARC-AGI tasks via epistemic exploration and program synthesis.  
  
    Architecture:  
    1. Explore: analyze training pairs to build a WorldModel  
    2. Synthesize: search for a CompositionalRule that satisfies all pairs  
    3. Verify: confirm the rule on training pairs with proof chain  
    4. Evolve: add successful rules to the ConceptLibrary for future tasks  
  
    All operations return ProofObjects. No neural components.  
    """  
  
    def __init__(self, max_depth: int = 3):  
        self.concept_library = ConceptLibrary()  
        self.max_depth = max_depth  
        self._init_library()  
  
    def _init_library(self) -> None:  
        """Seed the concept library with all single-op primitives."""  
        for op in PARAMETERLESS_OPS:  
            self.concept_library.primitives[op.value] = CompositionalRule(  
                [(op, {})]  
            )  
  
    def solve(self, task: ARCTask) -> Tuple[Solution, ProofObject]:  
        """  
        Main entry point. Returns (Solution, master_proof).  
        """  
        # Phase 1: Epistemic Exploration  
        world_model, explore_proof = self._explore(task)  
  
        # Phase 2: Program Synthesis  
        program, synthesis_proof = self._synthesize(task, world_model)  
  
        # Phase 3: Verification  
        verified, accuracy, verify_proof = self._verify(program, task)  
  
        # Phase 4: Library Evolution  
        evolve_proof = ProofObject(  
            "LibraryEvolution", ["no_new_concepts"], "Library unchanged"  
        )  
        if verified and program is not None:  
            evolve_proof = self._evolve(program)  
  
        # Master proof chain  
        all_proofs = [explore_proof, synthesis_proof, verify_proof, evolve_proof]  
        master_proof = ProofObject(  
            "ARCSolution",  
            [explore_proof, synthesis_proof, verify_proof, evolve_proof,  
             f"task_id={task.task_id}",  
             f"training_accuracy={accuracy}",  
             f"verified={verified}",  
             f"library_size={len(self.concept_library.compositions)}"],  
            f"ARC task {task.task_id}: {'solved' if verified else 'unsolved'}, "  
            f"accuracy={accuracy}, "  
            f"merkle={merkle_root_over_proofs(all_proofs)}",  
        )  
  
        solution = Solution(  
            program=program or Program(  
                rule=CompositionalRule([(PrimitiveOperation.IDENTITY, {})]),  
            ),  
            world_model=world_model,  
            training_accuracy=accuracy,  
            proof_chain=master_proof,  
        )  
        return solution, master_proof  
  
    def _explore(self, task: ARCTask) -> Tuple[WorldModel, ProofObject]:  
        """  
        Analyze training pairs to build a WorldModel.  
  
        Discovers:  
        - Whether the task requires conditional rules  
        - Grid property signatures across pairs  
        - Candidate mechanics (single-op rules that work on subsets)  
        """  
        world = WorldModel(grid_state=task.train_pairs[0][0])  
        interactions: List[Interaction] = []  
        discoveries: List[str] = []  
  
        # Check if conditional logic is needed  
        needs_conditional = requires_conditional(task.train_pairs)  
        interactions.append(Interaction(  
            action_type=InteractionType.OBSERVE,  
            observation=f"requires_conditional={needs_conditional}",  
        ))  
        if needs_conditional:  
            discoveries.append("conditional_branching_required")  
  
        # Analyze each pair for single-op matches  
        for i, (inp, out) in enumerate(task.train_pairs):  
            candidates = enumerate_single_ops(inp, out)  
            for c in candidates:  
                op_name = c.operations[0][0].value  
                if op_name not in world.mechanics:  
                    world.mechanics[op_name] = c  
                    discoveries.append(f"mechanic:{op_name}")  
            interactions.append(Interaction(  
                action_type=InteractionType.TEST_HYPOTHESIS,  
                observation=f"pair_{i}_candidates={len(candidates)}",  
            ))  
  
        # Build goal hypotheses  
        if world.mechanics:  
            for name, rule in world.mechanics.items():  
                score = sum(  
                    1 for inp, out in task.train_pairs  
                    if apply_rule(rule, inp) == out  
                ) / len(task.train_pairs)  
                world.goal_hypotheses.append(GoalHypothesis(  
                    description=f"apply_{name}",  
                    candidate_rule=rule,  
                    score=score,  
                ))  
  
        world.interaction_history = interactions  
  
        # Build epistemic model: worlds = possible rules, valuation = which pairs they satisfy  
        epistemic_worlds = set(world.mechanics.keys()) or {"unknown"}  
        epistemic_valuation = {}  
        for w in epistemic_worlds:  
            if w in world.mechanics:  
                rule = world.mechanics[w]  
                all_match = all(  
                    apply_rule(rule, inp) == out  
                    for inp, out in task.train_pairs  
                )  
                epistemic_valuation[w] = {"universal_rule": all_match}  
            else:  
                epistemic_valuation[w] = {"universal_rule": False}

SECTION 12: Known Limitations (to register as OI_ARC_001)
These are by design and documented in the ontology, not bugs:

Expressiveness ceiling. The bounded DSL over PrimitiveOperation cannot express transformations requiring unbounded iteration, recursive structure building, or novel primitives not in the enum. Tasks requiring these return (None, proof) with training_accuracy=0.0.

No neural transduction. The solver is purely symbolic. Tasks where the transformation is easier to learn statistically than to express compositionally will be missed.

Exploration is static. The _explore() phase analyzes training pairs but does not interact with an ARC-AGI-3 evaluation server. True interactive exploration requires an API client that is out of scope for this spec.

Concept library is session-scoped. The ConceptLibrary does not persist across Python processes. Cross-session transfer requires serialization (future work).

Depth-bounded search is exponential. enumerate_compositions() at max_depth=3 with 16 primitives generates up to 16³ = 4096 candidates per pair. At max_depth=4 this becomes 65,536. The max_depth parameter must be tuned per compute budget.

Each of these is tracked in OI_ARC_001 with resolution notes pointing to future primitive expansion, optional neural integration, server API client, library serialization, and pruning heuristics respectively.

That's the complete spec. Every function signature matches the actual ProofObject(rule, premises, conclusion) API. Every invariant test uses the existing register_invariant(AIInvariantTest(...)) / _claim() / verify_yeshua_standard() pipeline. Every ontology entry follows the existing JSON schemas.
