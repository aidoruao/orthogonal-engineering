# Session Log Truncation Report

## Affected File
`Kimi Code CLI na.txt` (commit 3391157)

## Symptom
The beginning of the session transcript is missing. The commit message
states: "ENTIRE BEGINNING HALF OF SESSION TRUNCATED FOR SOME REASON"

## Root Cause Analysis

Terminal emulators (GNOME Terminal, Windows Terminal, etc.) maintain a
**scrollback buffer** with a finite line limit (typically 1,000-10,000
lines). When a Kimi CLI session produces output exceeding this limit,
the oldest lines are discarded from the buffer.

When the user performs Ctrl+A (select all) → Ctrl+C (copy) at session
end, only the lines still in the scrollback buffer are captured.

## Prevention

### Option A: Increase scrollback buffer
```bash
# GNOME Terminal: Edit → Preferences → Profiles → Scrolling
# Set to "Unlimited" or 100,000 lines

# Windows Terminal: settings.json
# "historySize": 100000
```

### Option B: Use `script` command (recommended)
```bash
# Wrap Kimi CLI in script to capture ALL output
script -q "session_$(date +%Y%m%d_%H%M%S).txt" kimi

```

kimi 2>&1 | tee "session_$(date +%Y%m%d_%H%M%S).txt"

#!/usr/bin/env python3
"""CLI Session Usage Tracker — Glass-Box Audit Trail.

Logs Kimi CLI session metadata for sovereign audit trail.
All values are Fraction-based (no floats). Output is JSONL.

Usage:
    python3 tools/session_tracking/cli_usage_tracker.py \
        --session-id c223de88 \
        --context-used 169600 \
        --context-total 262100 \
        --domains-processed 10 \
        --commits 4
"""

import json
import hashlib
import argparse
from datetime import datetime, timezone
from pathlib import Path
from fractions import Fraction


def get_current_commit_hash() -> str:
    """Get HEAD commit hash."""
    import subprocess
    try:
        result = subprocess.run(
            ["git", "rev-parse", "HEAD"],
            capture_output=True, text=True, check=True
        )
        return result.stdout.strip()[:12]
    except Exception:
        return "unknown"


def log_session(
    session_id: str,
    context_used: int,
    context_total: int,
    domains_processed: int,
    commits: int,
    notes: str = "",
) -> dict:
    """Log a CLI session entry."""
    pct_num = context_used * 1000 // context_total
    pct = f"{pct_num // 10}.{pct_num % 10}%"

    entry = {
        "timestamp": datetime.now(timezone.utc).isoformat(),
        "session_id": session_id,
        "context_used": context_used,
        "context_total": context_total,
        "percentage": pct,
        "domains_processed": domains_processed,
        "commits": commits,
        "commit_hash": get_current_commit_hash(),
        "notes": notes,
        "entry_hash": "",
    }

    entry["entry_hash"] = hashlib.sha256(
        json.dumps(entry, sort_keys=True).encode()
    ).hexdigest()[:16]

    log_path = Path("logs/cli_usage.jsonl")
    log_path.parent.mkdir(exist_ok=True)
    with open(log_path, "a") as f:
        f.write(json.dumps(entry) + "\n")

    print(f"Logged: {session_id} ({pct} context, {domains_processed} domains, {commits} commits)")
    return entry


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Log CLI session usage")
    parser.add_argument("--session-id", required=True)
    parser.add_argument("--context-used", type=int, required=True)
    parser.add_argument("--context-total", type=int, default=262100)
    parser.add_argument("--domains-processed", type=int, default=0)
    parser.add_argument("--commits", type=int, default=0)
    parser.add_argument("--notes", default="")
    args = parser.parse_args()
    log_session(
        args.session_id, args.context_used, args.context_total,
        args.domains_processed, args.commits, args.notes
    )
TRACKEOF

        self.signifieds: List[Signified] = []

    def register_sign(self, sign: Sign) -> Tuple[Sign, ProofObject]:
        """Register a new sign with audit trail."""
        self.signs.append(sign)
        proof = ProofObject(
            conclusion=f"Sign registered: {sign.representamen} -> {sign.object_ref}",
            premises=[f"Kind: {sign.kind.name}", f"Universe: {sign.universe.name}"],
            rule="semiotic_registration",
            derivation=[f"Hash: {sign.hash}"]
        )
        return sign, proof

    def register_signifier(self, sf: Signifier) -> Tuple[Signifier, ProofObject]:
        """Register a signifier with etymology."""
        self.signifiers.append(sf)
        proof = ProofObject(
            conclusion=f"Signifier registered: {sf.form}",
            premises=[f"Domain: {sf.domain}", f"Etymology: {sf.etymology}"],
            rule="signifier_registration",
            derivation=["de Saussure: arbitrary but conventional"]
        )
        return sf, proof

    def register_signified(self, sd: Signified) -> Tuple[Signified, ProofObject]:
        """Register a signified with grounding."""
        self.signifieds.append(sd)
        proof = ProofObject(
            conclusion=f"Signified registered: {sd.concept}",
            premises=[f"Grounding: {sd.grounding}"],
            rule="signified_registration",
            derivation=[f"Falsifies if: {sd.falsifies_if}"]
        )
        return sd, proof

    def check_grounding(self, sign: Sign) -> Tuple[bool, ProofObject]:
        """Verify a sign is grounded (not floating/decorative)."""
        grounded = bool(sign.object_ref and sign.object_ref != "UNGROUNDED")
        proof = ProofObject(
            conclusion=f"Sign {'grounded' if grounded else 'UNGROUNDED'}: {sign.representamen}",
            premises=[f"Object ref: {sign.object_ref}"],
            rule="grounding_check",
            derivation=["A sign without an object is decoration"]
        )
        return grounded, proof

    def check_icon_isomorphism(self, sign_a: Sign, sign_b: Sign) -> Tuple[bool, ProofObject]:
        """Two iconic signs must be structurally similar."""
        iso = (sign_a.kind == SignKind.ICON and
               sign_b.kind == SignKind.ICON and
               sign_a.universe == sign_b.universe)
        proof = ProofObject(
            conclusion=f"Icon isomorphism {'holds' if iso else 'fails'}",
            premises=[f"A: {sign_a.representamen}", f"B: {sign_b.representamen}"],
            rule="icon_isomorphism",
            derivation=["Icons share structural form"]
        )
        return iso, proof

    def audit_all(self) -> Tuple[dict, ProofObject]:
        """Audit all registered signs for grounding."""
        grounded = 0
        ungrounded = 0
        for s in self.signs:
            g, _ = self.check_grounding(s)
            if g:
                grounded += 1
            else:
                ungrounded += 1
        total = len(self.signs)
        proof = ProofObject(
            conclusion=f"Semiotic audit: {grounded}/{total} grounded, {ungrounded} ungrounded",
            premises=[f"Total signs: {total}"],
            rule="semiotic_audit",
            derivation=[f"Grounding rate: {grounded}/{total}"]
        )
        return {"grounded": grounded, "ungrounded": ungrounded, "total": total}, proof

    total = len(glob.glob("src/domains/d_*/domain.py"))
    deepened = 0
    for inv in glob.glob("src/domains/d_*/invariants.py"):
        with open(inv) as f:
            if len(f.readlines()) >= 50:
                deepened += 1
    return {"total": total, "deepened": deepened, "stubs": total - deepened}


def count_axioms():
    """Count axiom modules."""
    return len(glob.glob("axioms/*.py"))


def count_case_studies():
    """Count case studies from index."""
    idx = "case_studies/CASE_STUDY_INDEX.json"
    if os.path.exists(idx):
        with open(idx) as f:
            data = json.load(f)
        return data.get("metadata", {}).get("total_complete", 0)
    return 0


def generate_csv(output_dir):
    """Generate CSV spreadsheets."""
    domains = count_domains()
    # Domain status CSV
    with open(os.path.join(output_dir, "domain_status.csv"), "w", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(["domain", "has_invariants", "line_count", "status"])
        for dp in sorted(glob.glob("src/domains/d_*/domain.py")):
            name = os.path.basename(os.path.dirname(dp))
            inv = os.path.join(os.path.dirname(dp), "invariants.py")
            if os.path.exists(inv):
                with open(inv) as fh:
                    lc = len(fh.readlines())
                writer.writerow([name, "yes", lc, "deepened" if lc >= 50 else "stub"])
            else:
                writer.writerow([name, "no", 0, "missing"])
    print(f"  CSV: {output_dir}/domain_status.csv")

    # Axiom coverage CSV
    with open(os.path.join(output_dir, "axiom_coverage.csv"), "w", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(["module", "line_count", "has_proofobject"])
        for ax in sorted(glob.glob("axioms/*.py")):
            with open(ax) as fh:
                content = fh.read()
                lc = len(content.splitlines())
                has_po = "ProofObject" in content
            writer.writerow([os.path.basename(ax), lc, has_po])
    print(f"  CSV: {output_dir}/axiom_coverage.csv")
