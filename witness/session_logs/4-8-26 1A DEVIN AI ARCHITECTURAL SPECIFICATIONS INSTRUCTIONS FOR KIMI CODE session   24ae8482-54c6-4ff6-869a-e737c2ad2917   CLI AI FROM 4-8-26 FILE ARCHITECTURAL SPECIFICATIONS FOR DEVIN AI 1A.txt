## Per-Case-Study Deliverables  
  
Each case study directory contains exactly 4 files:  
  
1. `gap_analysis.json` — Structured forensic analysis  
   - issue URL, repository, language, framework  
   - root cause (verbatim code quotes with file:line)  
   - invariant violations (which domain invariants are broken)  
   - affected components (list of files/classes/functions)  
   - fix proposal (code diff or pseudocode)  
   - falsification test (condition under which fix is wrong)  
   - SHA-256 hash of the analysis  
  
2. `pr_description.md` — Copy-paste ready GitHub comment  
   - Root Cause section (with code quotes)  
   - Fix section (with diff)  
   - Why This Works section  
   - Testing section  
   - Secular projection only  
  
3. `test_specification.md` — Falsification test cases  
   - Positive tests (fix works)  
   - Negative tests (original bug reproduced)  
   - Regression tests  
   - Performance benchmarks (if applicable)  
  
4. `ATTRIBUTION.md` — License, original authors, repo URL  
   - Original repo license  
   - Issue author attribution  
   - Non-affiliation statement  
   - Date of analysis

"""Validate all case study gap_analysis.json files against schema.  
  
Usage: python case_studies/framework/validate_case_studies.py  
"""  
import json  
import hashlib  
import sys  
from pathlib import Path  
  
def validate_case_study(path: Path) -> list:  
    """Validate a single gap_analysis.json file. Returns list of errors."""  
    errors = []  
    try:  
        data = json.loads(path.read_text())  
    except json.JSONDecodeError as e:  
        return [f"{path}: Invalid JSON: {e}"]  
  
    required = ["id", "issue_url", "repository", "language",  
                "analysis_date", "status", "domain_mapping",  
                "findings", "code_locations", "fix_proposal",  
                "falsification", "sha256_hash"]  
    for field in required:  
        if field not in data:  
            errors.append(f"{path}: Missing required field: {field}")  
  
    if "findings" in data:  
        for sub in ["summary", "root_cause", "affected_components"]:  
            if sub not in data["findings"]:  
                errors.append(f"{path}: findings missing: {sub}")  
  
    if "domain_mapping" in data:  
        for d in data["domain_mapping"]:  
            if not d.startswith("D_"):  
                errors.append(f"{path}: Invalid domain: {d}")  
  
    if "code_locations" in data:  
        for loc in data["code_locations"]:  
            if "file" not in loc or "line" not in loc:  
                errors.append(f"{path}: code_location missing file/line")  
  
    # Verify hash integrity  
    if "sha256_hash" in data:  
        content = json.dumps({k: v for k, v in sorted(data.items())  
                             if k != "sha256_hash"}, sort_keys=True)  
        expected = hashlib.sha256(content.encode()).hexdigest()  
        if data["sha256_hash"] != expected:  
            errors.append(f"{path}: Hash mismatch: "  
                         f"expected {expected[:16]}... "  
                         f"got {data['sha256_hash'][:16]}...")  
  
    return errors  
  
def main():  
    root = Path("case_studies")  
    if not root.exists():  
        print("ERROR: case_studies/ directory not found")  
        sys.exit(1)  
  
    all_errors = []  
    count = 0  
    for gap in root.rglob("gap_analysis.json"):  
        count += 1  
        errors = validate_case_study(gap)  
        all_errors.extend(errors)  
  
    print(f"Validated {count} case studies")  
    if all_errors:  
        for e in all_errors:  
            print(f"  ERROR: {e}")  
        sys.exit(1)  
    else:  
        print("All case studies valid")  
        sys.exit(0)  
  
if __name__ == "__main__":  
    main()

import os, glob  
  
domains = glob.glob('src/domains/d_*/invariants.py')

deepened = 0  
stubs = 0  
details = []  
  
for inv_path in sorted(domains):  
    domain = inv_path.split('/')[2]  
    with open(inv_path) as f:  
        content = f.read()  
    lines = len(content.strip().split('\n'))  
    funcs = content.count('def check_')  
    if lines >= 50 and funcs >= 2:  
        deepened += 1  
        status = 'DEEPENED'  
    else:  
        stubs += 1  
        status = 'STUB'  
    details.append((domain, lines, funcs, status))  
  
with open('DOMAIN_INVARIANT_STATUS.md', 'w') as f:  
    f.write('# Domain Invariant Status\n\n')  
    f.write(f'Generated: $(date -u +%Y-%m-%dT%H:%M:%SZ)\n\n')  
    f.write('## Summary\n\n')  
    f.write('| Metric | Count | Percentage |\n')  
    f.write('|--------|-------|------------|\n')  
    f.write(f'| Total domains | {total} | 100% |\n')

cat > civilizational_audit/framework/audit_entry.py << 'AUDITEOF'
"""Civilizational Audit Entry — structured complaint resolution.

Every entry is a falsifiable claim: "Complaint X is resolved by Y."
If Y does not actually resolve X, the entry fails its falsification test.

Standard: Yeshua
Version: 1.0.0
"""

from dataclasses import dataclass, field
from enum import Enum
from typing import List, Optional
from fractions import Fraction


class AuditStatus(Enum):
    RESOLVED = "RESOLVED"
    PARTIALLY_RESOLVED = "PARTIALLY_RESOLVED"
    CATALOGUED = "CATALOGUED"
    FALSIFIED = "FALSIFIED"


class SourceCategory(Enum):
    GAME_CRITICS = "GAME_CRITICS"
    TECH_ANALYSTS = "TECH_ANALYSTS"
    AI_CRITICS = "AI_CRITICS"
    INDUSTRY_OBSERVERS = "INDUSTRY_OBSERVERS"
    SECURITY_RESEARCHERS = "SECURITY_RESEARCHERS"
    LEGAL_ANALYSTS = "LEGAL_ANALYSTS"
    GAME_DESIGNERS = "GAME_DESIGNERS"
    OPEN_SOURCE = "OPEN_SOURCE"
    SUMMIT_PROCEEDINGS = "SUMMIT_PROCEEDINGS"
    ACADEMIC = "ACADEMIC"


@dataclass
class AuditSource:
    """Who said it, where, when."""
    name: str
    platform: str  # YouTube, Twitter, Blog, GDC, etc.
    url: str
    date: str  # ISO 8601
    category: SourceCategory
    subscriber_count: Optional[str] = None  # e.g. "1.2M"
    context: str = ""  # Video title, article title, etc.


@dataclass
class AuditEntry:
    """A single complaint-to-resolution mapping."""
    id: str  # e.g. "CA_GAME_001"
    source: AuditSource
    complaint: str  # Exact quote or faithful summary
    root_cause: str  # Ontological/architectural diagnosis
    resolution_domains: List[str]  # e.g. ["D_CONSUMER_PROTECTION", "D_GAMING"]
    resolution_axioms: List[str]  # e.g. ["axioms/measure_theory.py"]
    resolution_types: List[str]  # e.g. ["Contract", "Refinement"]
    resolution_description: str  # How OE resolves this
    falsification_test: str  # How to verify the resolution works
    status: AuditStatus = AuditStatus.CATALOGUED
    severity: str = "HIGH"  # HIGH / MEDIUM / LOW
    tags: List[str] = field(default_factory=list)


def validate_entry(entry: AuditEntry) -> bool:
    """Validate that an audit entry is complete and falsifiable."""
    assert entry.id, "Entry must have an ID"
    assert entry.complaint, "Entry must have a complaint"
    assert entry.root_cause, "Entry must have a root cause"
    assert entry.resolution_domains, "Entry must map to at least one domain"
    assert entry.falsification_test, "Entry must have a falsification test"
    assert entry.source.url, "Source must have a URL"
    return True
AUDITEOF
