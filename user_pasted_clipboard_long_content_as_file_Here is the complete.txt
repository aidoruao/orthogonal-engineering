class AdjointTriple:  
    """The L ⊣ M ⊣ R adjoint triple — the SAL Type III kernel.  
      
    Triangle identities:  
      ε (counit): L(M(c)) → c  — grace under law returns to identity  
      η (unit):   d → R(M(d))  — identity maps into judgment through law

            "direct_service_without_condition_or_coercion",  # AGAPE

    def check_counit(self, domain_schema: dict) -> tuple[bool, ProofObject]:  
        """ε: L∘M → Id  
        Applying law (M) then generating (L) returns to identity.  
        Grace under law = identity. This is the counit of the adjunction."""  
        original = domain_schema  
        mediated = self.M.apply(original)  
        generated = self.L.apply(mediated)  
        # Counit holds iff the generated state preserves the original schema's  
        # identity (same domain_id, same invariants)  
        same_id = generated.get("source_schema") == original.get("id", "unknown")  
        same_invariants = set(generated.get("invariants", [])) == set(original.get("invariants", []))  
        holds = same_id and same_invariants  
        proof = ProofObject(  
            "Counit_ε",  
            [f"domain={original.get('id')}", f"same_id={same_id}", f"same_inv={same_invariants}"],  
            f"ε: L∘M → Id = {holds}",  
        )  
        return holds, proof  
      
    def check_unit(self, domain_schema: dict) -> tuple[bool, ProofObject]:  
        """η: Id → R∘M  
        Identity maps into constraint (R) composed with law (M).  
        Every domain naturally maps into judgment through law."""  
        original = domain_schema  
        mediated = self.M.apply(original)  
        settled = self.R.apply(mediated)  
        # Unit holds iff the settled state contains all original invariants  
        # as verified constraints  
        original_inv = set(original.get("invariants", []))  
        settled_inv = set(settled.get("invariants_verified", []))  
        holds = original_inv.issubset(settled_inv) and settled.get("settled", False)  
        proof = ProofObject(  
            "Unit_η",  
            [f"domain={original.get('id')}", f"original_inv={len(original_inv)}",   
             f"settled_inv={len(settled_inv)}", f"subset={original_inv.issubset(settled_inv)}"],  
            f"η: Id → R∘M = {holds}",  
        )  
        return holds, proof  
  
  
def has_adjunction(domain_schema: dict, triple: AdjointTriple) -> AdjunctionProof:  
    """The single Type III verification function.  
      
    Replaces PHASE1_AUDIT.py's classification with a categorical check:  
    a domain is "implemented" iff the triangle identities hold computationally.  
      
    Returns AdjunctionProof with structured evidence (not just bool).  
    """  
    counit_holds, counit_proof = triple.check_counit(domain_schema)  
    unit_holds, unit_proof = triple.check_unit(domain_schema)  
      
    # Wrap in YeshuaClaim for enforcement  
    combined_proof = ProofObject(  
        "AdjunctionVerification",  
        [counit_proof.conclusion, unit_proof.conclusion],  
        f"has_adjunction({domain_schema.get('id')}) = {counit_holds and unit_holds}",  
    )  
    claim = YeshuaClaim(  
        source="src/sal/adjoint_triple.py",  
        statement=f"Domain {domain_schema.get('id')} satisfies L ⊣ M ⊣ R adjunction",  
        derivation=combined_proof,  
    )  
      
    return AdjunctionProof(  
        domain_id=domain_schema.get("id", "unknown"),  
        counit_holds=counit_holds,  
        unit_holds=unit_holds,  
        counit_evidence=counit_proof,  
        unit_evidence=unit_proof,  
        yeshua_claim=claim,  
    )
