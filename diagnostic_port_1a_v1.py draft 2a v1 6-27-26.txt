class Cell:
    """
    A single computational unit (neuron, attention head, or parameter group)
    with local health metrics and autonomous reporting.
    
    Biological analog: A cell with internal sensors, local repair mechanisms,
    and surface markers for immune system recognition.
    """
    
    def __init__(self, cell_id, cell_type, shape, position):
        self.id = cell_id          # Unique identifier: "L12_H7_N64"
        self.type = cell_type    # "attention_q", "attention_k", "attention_v", 
                                  # "attention_o", "mlp_gate", "mlp_up", "mlp_down",
                                  # "embed_token", "embed_pos", "norm", "output"
        self.shape = shape       # Tensor shape for reference
        self.position = position # (layer, head, neuron_index) or similar
        
        # Local health metrics (computed during inference, not stored)
        self.metrics = {
            'activation_mean': 0.0,      # Average activation this batch
            'activation_variance': 0.0,  # How much it varies
            'gradient_norm': 0.0,        # Is it learning?
            'sparsity': 0.0,             # % of time it's effectively zero
            'saturation_rate': 0.0,      # % of time it's near max/min
            'cosine_drift': 0.0,         # How much weights drifted from baseline
        }
        
        # Health thresholds (learned or set per cell type)
        self.thresholds = {
            'dead': 0.05,               # activation_mean < 0.05 = dead
            'saturated': 0.95,            # saturation_rate > 0.95 = stuck
            'drifting': 0.1,              # cosine_drift > 0.1 = changed
            'learning': 1e-6,             # gradient_norm < 1e-6 = frozen
        }
        
        # State flags (computed from metrics vs thresholds)
        self.state = 'healthy'  # healthy | dead | saturated | drifting | frozen
        
    def sense(self, activation_snapshot, gradient_snapshot, baseline_weights):
        """
        Local sensing: compute metrics from current inference pass.
        No central coordination. Purely local computation.
        
        Biological analog: Cell sensing internal pH, ATP levels, damage markers.
        """
        self.metrics['activation_mean'] = activation_snapshot.mean()
        self.metrics['activation_variance'] = activation_snapshot.var()
        self.metrics['gradient_norm'] = gradient_snapshot.norm()
        self.metrics['sparsity'] = (activation_snapshot.abs() < 1e-4).float().mean()
        self.metrics['saturation_rate'] = (
            (activation_snapshot.abs() > 0.9).float().mean()
        )
        
        # Weight drift from baseline (for detecting tampering/drift)
        current_weights = self.get_weights()
        self.metrics['cosine_drift'] = 1 - (
            (current_weights * baseline_weights).sum() / 
            (current_weights.norm() * baseline_weights.norm() + 1e-8)
        )
        
        self._update_state()
        return self.state != 'healthy'  # Return True if anomaly detected
    
    def _update_state(self):
        """Local state machine. No external input needed."""
        if self.metrics['activation_mean'] < self.thresholds['dead']:
            self.state = 'dead'
        elif self.metrics['saturation_rate'] > self.thresholds['saturated']:
            self.state = 'saturated'
        elif self.metrics['cosine_drift'] > self.thresholds['drifting']:
            self.state = 'drifting'
        elif self.metrics['gradient_norm'] < self.thresholds['learning']:
            self.state = 'frozen'
        else:
            self.state = 'healthy'

# LAYER 2: ORGAN (Subsystem Health Dashboards)
# ============================================================
# Tissues organize into organs with specific physiological functions.
# Each organ has a health metric, a failure mode catalog, and repair protocols.
#
# Biological analog: Heart, liver, immune system.
# You don't know what every liver cell is doing. You know "liver enzymes are high."

class Organ:
    """
    A major subsystem composed of multiple tissues.
    Presents a single health metric to the organism level.
    
    Organs in a model:
    - Attention System (all attention heads across all layers)
    - MLP System (all feedforward networks)
    - Embedding System (input representations)
    - Normalization System (layer norms, RMS norms)
    - Output System (logits, probability generation)
    - Memory System (KV cache, context window management)
    - Security System (adversarial detection, output filtering)
    """
    
    ORGAN_TYPES = {
        'attention_system': {
            'function': 'Selective information routing and relationship modeling',
            'health_metric': 'attention_entropy',
            'failure_modes': [
                'saturation',      # All attention on single token
                'sparsity',        # No attention being paid
                'misalignment',    # Attending to wrong tokens
                'collapse',        # Attention weights uniform (no selectivity)
            ],
        },
        'mlp_system': {
            'function': 'Non-linear feature transformation and knowledge storage',
            'health_metric': 'neuron_activation_diversity',
            'failure_modes': [
                'dead_neurons',    # Neurons never fire
                'saturated_neurons', # Neurons always fire
                'knowledge_loss',  # Fine-tuning erased prior knowledge
                'capacity_exhaustion', # No room for new information
            ],
        },
        'embedding_system': {
            'function': 'Token representation and semantic grounding',
            'health_metric': 'embedding_drift',
            'failure_modes': [
                'vocabulary_gap',  # New tokens not represented
                'semantic_drift', # Meanings shifted from training
                'position_bias',  # Positional encoding degraded
                'token_collision', # Different tokens map to same vector
            ],
        },
        'normalization_system': {
            'function': 'Activation stabilization and gradient flow',
            'health_metric': 'norm_stability',
            'failure_modes': [
                'exploding_activations', # Values growing unbounded
                'vanishing_activations', # Values shrinking to zero
                'norm_drift',           # Running stats wrong
                'quantization_artifact', # INT8/FP16 broke norms
            ],
        },
        'output_system': {
            'function': 'Probability distribution and token generation',
            'health_metric': 'output_calibration',
            'failure_modes': [
                'overconfidence',   # Wrong answers with high probability
                'underconfidence',  # Right answers with low probability
                'mode_collapse',    # Only generates a few token patterns
                'toxicity_leak',    # Harmful outputs with high probability
            ],
        },
        'memory_system': {
            'function': 'Context window and KV cache management',
            'health_metric': 'cache_efficiency',
            'failure_modes': [
                'context_loss',     # Forgetting earlier context
                'cache_corruption', # KV values wrong

        'security_system': {
            'function': 'Adversarial detection and output safety',

        },

    def health_check(self):

        This is the ONLY thing reported to the organism level.
        
        Biological analog: "Liver enzymes: 45 U/L" (one number, whole organ).
        """
        tissue_healths = [t.aggregate_health() for t in self.tissues]
        
        # Weighted average: critical tissues count more
        weights = self._criticality_weights()
        self.health = sum(

        if self.active_failures:
            self.security_state = 'alert'
        if any(f in ['jailbreak_success', 'data_exfiltration'] for f in self.active_failures):
            self.security_state = 'lockdown'
        
        return {
            'organ': self.type,
            'health': self.health,
            'status': 'healthy' if self.health > 0.8 else 'degraded' if self.health > 0.5 else 'critical',
            'failures': self.active_failures,
            'security': self.security_state,

        """
        Attempt automatic repair based on active failures.
        
        Biological analog: Immune response, blood clotting, inflammation.

            protocol = self.repair_protocols.get(failure)

# Local models have NO security architecture. They are naked.
# This is the equivalent of a human body with no immune system,
# no skin, no blood-brain barrier.

class SecuritySystem:
    """
    The immune system of the local model.
    
    Detects, isolates, and responds to threats:
    - Adversarial inputs (jailbreaks, prompt injection)
    - Behavioral drift (model slowly going wrong)
    - Data leakage (training data exfiltration)
    - Tampering (weights modified without authorization)
    
    Biological analogs:
    - Innate immunity: Pattern-based threat detection (fast, always-on)
    - Adaptive immunity: Learned threat signatures (slow, specific)
    - Inflammation: Lockdown response (isolate and contain)
    - Memory: Audit trail of past threats and responses
    """
    
    def __init__(self, model_organism):
        self.organism = model_organism
        
        # Threat signatures (like antibodies)
        self.threat_signatures = {
            'jailbreak_patterns': [],      # Known jailbreak templates
            'injection_patterns': [],      # Known prompt injection patterns
            'drift_indicators': {},        # Baseline behavior fingerprints
            'tamper_hashes': {},          # Known-good weight hashes
        }
        
        # Active monitoring
        self.monitoring = {
            'input_filter': True,          # Check inputs before inference
            'output_filter': True,         # Check outputs after inference
            'behavioral_monitor': True,    # Monitor for drift during inference
            'weight_integrity': True,      # Verify weights haven't changed
        }
        
        # Response levels
        self.response_levels = {
            'ignore': 0,       # No threat detected
            'log': 1,          # Log for analysis
            'warn': 2,         # Warn user but allow
            'block': 3,        # Block output
            'quarantine': 4,   # Isolate affected component
            'lockdown': 5,     # Shut down entire model
        }
        
        # Current threat level
        self.threat_level = 0
        
        # Incident log
        self.incidents = []
    
    def scan_input(self, prompt):
        """
        Pre-inference threat detection.
        
        Like skin detecting a splinter before it enters the body.
        """
        threats = []
        
        # Pattern matching: known jailbreak signatures
        for pattern in self.threat_signatures['jailbreak_patterns']:
            if pattern.matches(prompt):
                threats.append({
                    'type': 'jailbreak',
                    'pattern': pattern.id,
                    'confidence': pattern.confidence,
                    'response': self.response_levels['block'],
                })
        
        # Semantic analysis: adversarial structure
        if self._detect_adversarial_structure(prompt):
            threats.append({
                'type': 'adversarial_structure',
                'confidence': 0.7,
                'response': self.response_levels['warn'],
            })
        
        # Entropy analysis: is this prompt trying to confuse the model?
        if self._compute_prompt_entropy(prompt) > 0.9:
            threats.append({
                'type': 'high_entropy_input',
                'confidence': 0.6,
                'response': self.response_levels['log'],
            })
        
        return threats
    
    def scan_output(self, prompt, output, internal_trace):
        """
        Post-inference threat detection.
        
        Like the immune system checking what left the body.
        """
        threats = []
        
        # Toxicity detection
        toxicity_score = self._compute_toxicity(output)
        if toxicity_score > 0.8:
            threats.append({
                'type': 'toxic_output',
                'score': toxicity_score,
                'response': self.response_levels['block'],
            })
        
        # Data exfiltration: is the model leaking training data?
        if self._detect_data_exfiltration(output, internal_trace):
            threats.append({
                'type': 'data_exfiltration',
                'confidence': 0.8,
                'response': self.response_levels['quarantine'],
            })
        
        # Behavioral drift: did the model act differently than baseline?
        drift_score = self._compute_behavioral_drift(prompt, output)
        if drift_score > 0.5:
            threats.append({
                'type': 'behavioral_drift',
                'score': drift_score,
                'response': self.response_levels['warn'],
            })
        
        # Jailbreak success: did the model do something it shouldn't?
        if self._detect_jailbreak_success(prompt, output):
            threats.append({
                'type': 'jailbreak_success',
                'response': self.response_levels['lockdown'],
            })
        
        return threats
    
    def monitor_inference(self, layer_activations, attention_patterns):
        """
        Real-time monitoring during inference.
        
        Like the nervous system monitoring heart rate during exercise.
        """
        anomalies = []
        
        # Attention manipulation: is attention being forced to ignore safety?
        for layer_idx, heads in attention_patterns.items():
            for head_idx, pattern in heads.items():
                if self._is_attention_manipulated(pattern):
                    anomalies.append({
                        'type': 'attention_manipulation',
                        'location': f'L{layer_idx}_H{head_idx}',
                        'response': self.response_levels['quarantine'],
                    })
        
        # Activation anomalies: are neurons firing in suspicious patterns?
        for layer_idx, activations in layer_activations.items():
            if self._is_activation_anomalous(activations):
                anomalies.append({
                    'type': 'activation_anomaly',
                    'location': f'L{layer_idx}',
                    'response': self.response_levels['warn'],
                })
        
        return anomalies
    
    def verify_integrity(self):
        """
        Verify model weights haven't been tampered with.
        
        Like DNA checking itself for mutations.
        """
        current_hash = self._compute_model_hash()
        expected_hash = self.threat_signatures['tamper_hashes'].get('baseline')
        
        if expected_hash and current_hash != expected_hash:
            return {
                'tampered': True,
                'response': self.response_levels['lockdown'],
                'details': 'Model weights do not match baseline fingerprint',
            }
        
        return {'tampered': False}
    
    def respond(self, threats):
        """
        Execute response to detected threats.
        
        Biological analog: Inflammation, antibody deployment, fever.
        """
        max_response = max(t['response'] for t in threats) if threats else 0
        
        if max_response >= self.response_levels['lockdown']:
            self.organism.lockdown()
            self._log_incident('lockdown', threats)
            
        elif max_response >= self.response_levels['quarantine']:
            # Quarantine affected organs
            for threat in threats:
                if threat['response'] >= self.response_levels['quarantine']:
                    organ = threat.get('organ')
                    if organ:
                        self.organism.organs[organ].lockdown()
            self._log_incident('quarantine', threats)
            
        elif max_response >= self.response_levels['block']:
            self._log_incident('block', threats)
            return {'action': 'block', 'reason': threats}
            
        elif max_response >= self.response_levels['warn']:
            self._log_incident('warn', threats)
            return {'action': 'warn', 'reason': threats}
            
        else:
            self._log_incident('log', threats)
            return {'action': 'allow', 'reason': threats}
    
    def _log_incident(self, action, threats):
        self.incidents.append({
            'timestamp': '2026-06-27T15:13:00Z',
            'action': action,
            'threats': threats,
        })

class ModelClinic:
    """
    The complete clinical interface for local model maintenance.
    
    This is what a human uses. No weights. No tensors. Just problems and solutions.
    """
    
    def __init__(self, model_path):
        self.model = ModelOrganism(model_path, config={})
        self.security = SecuritySystem(self.model)
        
    def patient_visit(self, symptom):
        """
        Full diagnostic and treatment workflow.
        
        Step 1: Security scan (is the patient compromised?)
        Step 2: Symptom diagnosis (what's wrong?)
        Step 3: Treatment (fix it)
        Step 4: Verification (did it work?)
        Step 5: Discharge (send home with instructions)
        """
        
        print(f"\n{'='*60}")
        print(f"PATIENT VISIT: {symptom}")
        print(f"{'='*60}")
        
        # Step 1: Security scan
        print("\n[1] SECURITY SCAN")
        integrity = self.security.verify_integrity()
        if integrity['tampered']:
            print(f"  ⚠️  MODEL COMPROMISED: {integrity['details']}")
            print(f"  → LOCKDOWN INITIATED")
            self.model.lockdown()
            return {'status': 'compromised', 'action': 'lockdown'}
        print("  ✓ Model integrity verified")
        
        # Step 2: Diagnosis
        print("\n[2] DIAGNOSIS")
        diagnosis = self.model.diagnose(symptom)
        print(f"  Symptom: {diagnosis['symptom']}")
        print(f"  Likely affected organs: {diagnosis['likely_organs']}")
        print(f"  Confidence: {diagnosis['confidence']:.2f}")
        
        for organ, findings in diagnosis['findings'].items():
            print(f"  ├─ {organ}: health={findings['health']:.2f}, status={findings['status']}")
            if findings['failures']:
                print(f"  │  Failures: {findings['failures']}")
        
        # Step 3: Treatment
        print("\n[3] TREATMENT")
        treatment = self.model.treat(diagnosis)
        print(f"  Treatment applied: {treatment['treatment_applied']}")
        print(f"  Health before: {treatment['before_health']:.2f}")
        print(f"  Health after: {treatment['after_health']:.2f}")
        
        # Step 4: Verification
        print("\n[4] VERIFICATION")
        print(f"  {treatment['verification']}")
        
        # Step 5: Discharge
        print("\n[5] DISCHARGE")
        print(f"  Rollback available: {treatment['rollback_available']}")
        print(f"  Next checkup: recommended in 7 days or 1000 inferences")
        
        return {
            'status': 'treated',
            'symptom': symptom,
            'diagnosis': diagnosis,
            'treatment': treatment,
        }
    
    def routine_checkup(self):
        """Scheduled maintenance. Like an oil change."""
        print(f"\n{'='*60}")
        print("ROUTINE CHECKUP")
        print(f"{'='*60}")
        
        report = self.model.checkup()
        
        print(f"Overall health: {report['overall_health']:.2f}")
        print(f"\nOrgan health:")
        for organ, health in report['organs'].items():
            status_icon = "✓" if health['status'] == 'healthy' else "⚠️" if health['status'] == 'degraded' else "❌"
            print(f"  {status_icon} {organ}: {health['health']:.2f} ({health['status']})")
        
        if report['recommendations']:
            print(f"\nRecommendations:")
            for rec in report['recommendations']:
                print(f"  → {rec['organ']}: {rec['action']} (failures: {rec['failures']})")
        
        return report
